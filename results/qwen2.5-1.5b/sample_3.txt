import torch
import torch.nn as nn
import torch.nn.functional as F

def build_model(rpm: float) -> nn.Module:
    class BearingModel(nn.Module):
        def __init__(self, rpm):
            super(BearingModel, self).__init__()
            # Constants
            self.rpm = rpm
            self.diameter = 0.3126
            self.pitch_diameter = 1.537
            self.contact_angle = 0

            # Convolutional layers
            self.conv1 = nn.Conv1d(1, 16, 3, padding=1)
            self.relu1 = nn.ReLU()
            self.pool1 = nn.MaxPool1d(2)
            self.conv2 = nn.Conv1d(16, 32, 3, padding=1)
            self.relu2 = nn.ReLU()
            self.pool2 = nn.MaxPool1d(2)

            # Fully connected layers
            self.fc1 = nn.Linear(32 * 1200, 100)
            self.relu3 = nn.ReLU()
            self.fc2 = nn.Linear(100, 4)

        def forward(self, x):
            x = self.conv1(x)
            x = self.relu1(x)
            x = self.pool1(x)
            x = self.conv2(x)
            x = self.relu2(x)
            x = self.pool2(x)
            x = x.view(x.size(0), -1)
            x = self.fc1(x)
            x = self.relu3(x)
            x = self.fc2(x)
            return x

    return BearingModel(rpm)
