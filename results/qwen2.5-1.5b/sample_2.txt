```python
import torch
import torch.nn as nn
import torch.nn.functional as F

# Define the model
class BearingDiagnosisModel(nn.Module):
    def __init__(self, rpm):
        super(BearingDiagnosisModel, self).__init__()
        self.rpm = rpm
        self.conv1 = nn.Conv1d(1, 32, kernel_size=3, stride=1)
        self.conv2 = nn.Conv1d(32, 64, kernel_size=3, stride=1)
        self.fc1 = nn.Linear(64 * 24000, 512)
        self.fc2 = nn.Linear(512, 4)

    def forward(self, x):
        x = F.relu(self.conv1(x))
        x = F.relu(self.conv2(x))
        x = x.view(-1, 64 * 24000)
        x = F.relu(self.fc1(x))
        x = self.fc2(x)
        return x

# Build the model
def build_model(rpm):
    return BearingDiagnosisModel(rpm)

# Example usage
if __name__ == "__main__":
    model = build_model(1500.0)
    print(model)
```
