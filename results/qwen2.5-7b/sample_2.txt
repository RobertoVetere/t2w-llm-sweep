        # Example weights for the first convolutional layer

        return np.random.rand(16, 1, 16) * 0.1 - 0.05

        return np.random.rand(32, 16, 16) * 0.1 - 0.05

        # Example weights for the first fully connected layer
        return np.random.rand(128, 32 * (self.num_samples - 31)) * 0.1 - 0.05

        # Example bias for the first fully connected layer
        return np.zeros(128)

        # Example bias for the second fully connected layer
