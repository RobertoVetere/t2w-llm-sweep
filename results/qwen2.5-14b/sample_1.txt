import torch
import numpy as np

def build_model(rpm: float) -> torch.nn.Module:
    # SKF 6205-2RS bearing parameters
    ball_diameter = 0.3126  # inches
    pitch_diameter = 1.537  # inches
    num_balls = 9

    # Calculate the frequency of the fault features
    # Shaft speed in Hz
    shaft_speed_hz = rpm / 60.0
    # Ball passing frequency (BPFO)
    bpfo = (shaft_speed_hz * num_balls) / pitch_diameter
    # Ball passing frequency (BPFI)
    bpfi = shaft_speed_hz * num_balls
    # Ball spin frequency (BSF)
    bsf = (2 * shaft_speed_hz * ball_diameter) / pitch_diameter
    # Fundamental train frequency (FTF)
    ftf = shaft_speed_hz / pitch_diameter

    # Frequencies for fault diagnosis
    frequencies = [bpfo, bpfi, bsf, ftf]

    # Create a simple 1D CNN model to diagnose bearing faults
    class BearingDiagnosisModel(torch.nn.Module):
        def __init__(self):
            super(BearingDiagnosisModel, self).__init__()
            # Input shape: (batch, 1, 24000)
            self.conv1 = torch.nn.Conv1d(in_channels=1, out_channels=32, kernel_size=3, stride=1, padding=1)
            self.pool = torch.nn.MaxPool1d(kernel_size=2, stride=2, padding=0)
            self.conv2 = torch.nn.Conv1d(in_channels=32, out_channels=64, kernel_size=3, stride=1, padding=1)
            self.fc1 = torch.nn.Linear(in_features=64 * 3000, out_features=128)
            self.fc2 = torch.nn.Linear(in_features=128, out_features=4)  # 4 output classes
            self.relu = torch.nn.ReLU()

            # Set weights based on frequencies
            for param in self.parameters():
                if 'weight' in param.name:
                    # Assuming the weights are related to the frequencies in a linear manner
                    # This is a placeholder for the actual frequency-based weights
                    param.data = torch.tensor([[frequencies[0]], [frequencies[1]], [frequencies[2]], [frequencies[3]]], dtype=torch.float)

        def forward(self, x):
            x = self.pool(self.relu(self.conv1(x)))
            x = self.pool(self.relu(self.conv2(x)))
            x = x.view(-1, 64 * 3000)
            x = self.relu(self.fc1(x))
            x = self.fc2(x)
            return x

    return BearingDiagnosisModel()

# Example usage
model = build_model(1800)
